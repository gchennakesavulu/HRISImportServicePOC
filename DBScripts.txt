CREATE PROCEDURE [dbo].[MergeJobTemplateData]
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRANSACTION;

    BEGIN TRY

        /* Merge JobTemplate */
        MERGE Admin.JobTemplate AS T
        USING dbo.Stg_JobTemplate AS S
        ON T.Code = S.Code

        WHEN MATCHED THEN
            UPDATE SET
                T.Name = S.Name,
				T.JobPostingTitle = S.Name,
                T.JobSummary = S.JobDescription,
                T.Hours = S.Hours,
                T.CategoryID = S.CategoryID,
                T.EEOJobGroupID = S.EEOJobGroupID,
                T.ScheduleID = S.ScheduleID,
                T.ShiftID = S.ShiftID,
                T.SalaryRangeID = S.SalaryRangeID,
                T.PayGradeID = S.PayGradeID,
                T.JobStatusID = S.JobStatusID,
                T.HRISIdentifier = S.HRISIdentifier,
				T.CreateDate = GETUTCDATE(),
                T.ModifiedDate = GETUTCDATE(),
				T.CreatedBy=5,
				T.Modifiedby=5

        WHEN NOT MATCHED BY TARGET THEN
            INSERT
            (
                Name,
				JobPostingTitle,
                Code,
                JobSummary,
                Hours,
                CategoryID,
                EEOJobGroupID,
                ScheduleID,
                ShiftID,
                SalaryRangeID,
                PayGradeID,
                JobStatusID,
                HRISIdentifier,
                Active,
                IsAvailableInJobAgent,
                Deleted,
                CreateDate,
				ModifiedDate,
                CreatedBy,
				Modifiedby
            )
            VALUES
            (
                S.Name,
				S.Name,
                S.Code,
                S.JobDescription,
                S.Hours,
                S.CategoryID,
                S.EEOJobGroupID,
                S.ScheduleID,
                S.ShiftID,
                S.SalaryRangeID,
                S.PayGradeID,
                S.JobStatusID,
                S.HRISIdentifier,
                S.Active,
                S.IsAvailableInJobAgent,
                S.Deleted,
                S.CreateDate,
				S.ModifiedDate,
                S.CreatedBy,
				S.Modifiedby
            );

        /* Merge JobTemplateBusinessEntity */
        MERGE Admin.JobTemplateBusinessEntity AS T
        USING
        (
            SELECT
                JT.JobTemplateID,
                SBE.BusinessEntityID,
                SBE.Deleted,
                SBE.CreateDate,
				SBE.ModifiedDate,
                SBE.CreatedBy,              
                SBE.ModifiedBy
            FROM dbo.Stg_JobTemplateBusinessEntity SBE
            INNER JOIN dbo.Stg_JobTemplate ST
                ON ST.Code = SBE.Code
            INNER JOIN admin.JobTemplate JT
                ON JT.Code = ST.Code
        ) AS S
        ON T.JobTemplateID = S.JobTemplateID
           AND T.BusinessEntityID = S.BusinessEntityID

        WHEN MATCHED THEN
            UPDATE SET
                T.ModifiedDate = GETUTCDATE(),
                T.ModifiedBy = S.ModifiedBy

        WHEN NOT MATCHED BY TARGET THEN
            INSERT
            (
                JobTemplateID,
                BusinessEntityID,
                Deleted,
                CreateDate,
				ModifiedDate,
                CreatedBy,
				ModifiedBy
            )
            VALUES
            (
                S.JobTemplateID,
                S.BusinessEntityID,
                S.Deleted,
                S.CreateDate,
			     GETUTCDATE(),
                S.CreatedBy,
				5

            );

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH
END
GO

CREATE PROCEDURE dbo.ClearJobTemplateStageTables
AS
BEGIN
    SET NOCOUNT ON;

    BEGIN TRY
        -- Clear staging tables
        DELETE FROM dbo.Stg_JobTemplateBusinessEntity;
        DELETE FROM dbo.Stg_JobTemplate;

        PRINT 'Staging tables cleared successfully.';
    END TRY
    BEGIN CATCH
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrorSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrorState INT = ERROR_STATE();

        RAISERROR(@ErrorMessage, @ErrorSeverity, @ErrorState);
    END CATCH
END;
